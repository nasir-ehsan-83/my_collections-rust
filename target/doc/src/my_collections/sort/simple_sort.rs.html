1/// Sorts a array in-place using the insertion sort algorithm.
2///
3/// Insertion sort is $O(n^2)$ in the average and worst cases, but $O(n)$ 
4/// for nearly-sorted data. It is efficient for very small datasets.
5///
6/// # Examples
7///
8/// ```
9/// let mut numbers = [4, 2, 5, 1, 3];
10/// my_collections::sort::insertion_sort(&mut numbers);
11/// assert_eq!(numbers, [1, 2, 3, 4, 5]);
12/// ```
13
14pub fn insertion_sort<T: PartialOrd + Clone>(array: &mut [T]) {
15    // 1. We start at index 1 because the first element is "sorted" by itself
16    for i in 1..array.len() {
17        // 2. "Pick up" the current element (this is the Key)
18        let key = array[i].clone();
19        let mut j = i;
20
21        // 3. Shift elements to the right as long as they are larger than the key
22        // We use 'j > 0' to prevent going out of bounds
23        while j > 0 && array[j - 1] > key {
24            array[j] = array[j - 1].clone(); // Shift the larger element right
25            j -= 1; // Move the "hole" to the left
26        }
27
28        // 4. Drop the key into the final "hole"
29        array[j] = key;
30    }
31}
